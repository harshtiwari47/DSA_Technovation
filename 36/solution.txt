class Solution {
public:
    bool isValidSudoku(vector<vector<char>>& board) {
        int n = board.size();
        int m = board[0].size();

        vector<unordered_set<char>> rows(n);
        vector<unordered_set<char>> cols(m);
        vector<unordered_set<char>> grid(9);

        for (int i = 0; i < n; i++) {
            for (int j = 0; j < m; j++) {
                if (board[i][j] == '.')
                    continue;

                if (rows[i].count(board[i][j])) {
                    return false;
                }

                if (cols[j].count(board[i][j])) {
                    return false;
                }

                int gridIndex = (i / 3) * 3 + (j / 3);

                if (grid[gridIndex].count(board[i][j])) {
                    return false;
                }

                rows[i].insert(board[i][j]);
                cols[j].insert(board[i][j]);
                grid[gridIndex].insert(board[i][j]);
            }
        }

        return true;
    }
};
