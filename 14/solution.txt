class Solution {
public:
    string longestCommonPrefix(vector<string>& strs) {
        string prefix = strs[0];
        for (const auto str : strs) {
            if (str[0] != prefix[0])
                return "";

            for (int i = 0; i < str.length(); i++) {
                if (prefix[i] != str[i]) {
                    prefix = prefix.substr(0, i);
                    continue;
                }
            }

            if (str.length() < prefix.length())
                prefix = str;
        }

        return prefix;
    }
};
